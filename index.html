import json import math import os import random import pygame # Khởi tạo Pygame pygame.init() pygame.mixer.init() # Cấu hình màn hình SCREEN_WIDTH = 900 SCREEN_HEIGHT = 600 screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("Game Nâng Cấp Item - Kiếm Hiệp RPG") clock = pygame.time.Clock() # Màu sắc WHITE = (255, 255, 255) BLACK = (0, 0, 0) RED = (220, 50, 50) GREEN = (50, 200, 50) GOLD = (255, 215, 0) DARK_RED = (120, 20, 20) WOOD_BG = (139, 69, 19) WOOD_BORDER = (92, 51, 23) UI_BG = (30, 30, 35, 220) # Trạng thái game # 'LOGIN', 'PLAYING', 'UPGRADE' game_state = "LOGIN" # Dữ liệu người chơi current_user = "" player_data = {"gold": 1000, "items": 39} # Input đăng nhập username_input = "" password_input = "" active_box = "user" # 'user' hoặc 'pass' login_message = "" # Font chữ (Sử dụng font mặc định hệ thống hỗ trợ tiếng Việt cơ bản) font_path = None try: font_small = pygame.font.SysFont("Arial", 16) font_medium = pygame.font.SysFont("Arial", 20, bold=True) font_large = pygame.font.SysFont("Arial", 28, bold=True) except: font_small = pygame.font.Font(None, 24) font_medium = pygame.font.Font(None, 28) font_large = pygame.font.Font(None, 36) # Tạo âm thanh nền đơn giản bằng code (Synthesized background music/beeps) try: sample_rate = 22050 duration = 1.5 t = [i / sample_rate for i in range(int(sample_rate * duration))] # Tạo giai điệu nhẹ nhàng buffer = [ int( 32767 * 0.3 * math.sin(2 * math.pi * 220 * x) * (1 - x / duration) ) for x in t ] sound_array = pygame.sndarray.make_sound( bytes(bytearray([b for s in buffer for b in s.to_bytes(2, 'little', signed=True)])) ) sound_array.play(loops=-1) except: pass # Nếu lỗi driver âm thanh thì bỏ qua def save_game_data(): if not current_user: return data = {} if os.path.exists("save_game.json"): try: with open("save_game.json", "r", encoding="utf-8") as f: data = json.load(f) except: pass data[current_user] = { "password": password_input, "gold": player_data["gold"], "items": player_data["items"], } with open("save_game.json", "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=4) def load_game_data(username, password): global player_data if not os.path.exists("save_game.json"): return False, "Chưa có tài khoản nào được lưu!" try: with open("save_game.json", "r", encoding="utf-8") as f: data = json.load(f) if username in data: if data[username]["password"] == password: player_data["gold"] = data[username].get("gold", 1000) player_data["items"] = data[username].get("items", 39) return True, "Đăng nhập thành công!" else: return False, "Sai mật khẩu!" else: # Tạo mới luôn nếu chưa có player_data["gold"] = 1000 player_data["items"] = 39 return True, "Tài khoản mới đã được tạo!" except: return False, "Lỗi đọc dữ liệu lưu trữ!" # Đối tượng Người chơi (Nhân vật áo đỏ ngầu) class Player: def __init__(self, x, y): self.x = x self.y = y self.speed = 4 self.width = 40 self.height = 60 def handle_keys(self): keys = pygame.key.get_pressed() if keys[pygame.K_LEFT] or keys[pygame.K_a]: self.x -= self.speed if keys[pygame.K_RIGHT] or keys[pygame.K_d]: self.x += self.speed if keys[pygame.K_UP] or keys[pygame.K_w]: self.y -= self.speed if keys[pygame.K_DOWN] or keys[pygame.K_s]: self.y += self.speed # Giới hạn trong màn hình self.x = max(50, min(SCREEN_WIDTH - 50, self.x)) self.y = max(100, min(SCREEN_HEIGHT - 60, self.y)) def draw(self, surface): # Vẽ mô phỏng nhân vật phong cách cổ trang áo đỏ (như hình mẫu) # Thân áo choàng đỏ pygame.draw.rect( surface, RED, (self.x - 15, self.y - 10, 30, 45), border_radius=5 ) # Viền đen trang trí áo pygame.draw.rect( surface, BLACK, (self.x - 15, self.y - 10, 30, 45), 2, border_radius=5 ) # Tóc đen dài & dải lụa đỏ phía sau pygame.draw.circle(surface, BLACK, (self.x, self.y - 20), 14) pygame.draw.rect(surface, DARK_RED, (self.x - 3, self.y - 30, 6, 25)) # Kiếm bên hông pygame.draw.line(surface, WHITE, (self.x + 12, self.y), (self.x + 22, self.y + 15), 3) # Đối tượng NPC Nâng Cấp (Kiếm khách kỳ bí áo đỏ) class NPC: def __init__(self, x, y): self.x = x self.y = y self.width = 50 self.height = 60 def draw(self, surface): # Vẽ NPC phong cách kiếm hiệp cổ trang đẹp mắt # Áo choàng đỏ đậm pygame.draw.rect( surface, DARK_RED, (self.x - 18, self.y - 10, 36, 48), border_radius=6 ) pygame.draw.rect( surface, GOLD, (self.x - 18, self.y - 10, 36, 48), 2, border_radius=6 ) # Tóc đen pygame.draw.circle(surface, BLACK, (self.x, self.y - 22), 15) # Hiệu hào quang/nón hoặc điểm nhấn kiếm hiệp pygame.draw.circle(surface, GOLD, (self.x, self.y - 32), 4) # Tên NPC name_text = font_small.render("NPC Lão Kiếm Sư", True, GOLD) surface.blit( name_text, (self.x - name_text.get_width() // 2, self.y - 55) ) player = Player(200, 300) npc = NPC(700, 350) # Biến thông báo kết quả nâng cấp upgrade_msg = "" upgrade_color = WHITE # Vòng lặp chính của game running = True while running: screen.fill((25, 25, 35)) # Màu nền bản đồ tối cổ trang for event in pygame.event.get(): if event.type == pygame.QUIT: save_game_data() running = False if game_state == "LOGIN": if event.type == pygame.MOUSEBUTTONDOWN: # Kiểm tra click vào ô nhập liệu if 300 <= event.pos[0] <= 600 and 220 <= event.pos[1] <= 260: active_box = "user" elif 300 <= event.pos[0] <= 600 and 300 <= event.pos[1] <= 340: active_box = "pass" # Click nút đăng nhập elif 350 <= event.pos[0] <= 550 and 380 <= event.pos[1] <= 430: if username_input.strip() and password_input.strip(): success, msg = load_game_data(username_input, password_input) login_message = msg if success: current_user = username_input game_state = "PLAYING" else: login_message = "Vui lòng nhập đầy đủ tên và mật khẩu!" elif event.type == pygame.KEYDOWN: if event.key == pygame.K_RETURN: if username_input.strip() and password_input.strip(): success, msg = load_game_data(username_input, password_input) login_message = msg if success: current_user = username_input game_state = "PLAYING" elif event.key == pygame.K_BACKSPACE: if active_box == "user": username_input = username_input[:-1] else: password_input = password_input[:-1] else: if event.unicode.isprintable(): if active_box == "user" and len(username_input) < 15: username_input += event.unicode elif active_box == "pass" and len(password_input) < 15: password_input += event.unicode elif game_state == "PLAYING": if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: save_game_data() game_state = "LOGIN" # Kiểm tra khoảng cách đến NPC để mở bảng nâng cấp dist = math.hypot(player.x - npc.x, player.y - npc.y) if dist < 60: game_state = "UPGRADE" upgrade_msg = "Hãy chọn nâng cấp trang bị!" elif game_state == "UPGRADE": if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE or event.key == pygame.K_x: game_state = "PLAYING" elif event.type == pygame.MOUSEBUTTONDOWN: mx, my = event.pos # Nút Nâng cấp (50% thành công, thất bại -1 item) if 375 <= mx <= 525 and 320 <= my <= 365: if player_data["gold"] >= 200 and player_data["items"] > 0: player_data["gold"] -= 200 if random.random() < 0.5: # Tỉ lệ thành công 50% player_data["items"] += 1 # Giả định thành công tăng cấp độ item upgrade_msg = "Nâng cấp thành công! +1 cấp Item!" upgrade_color = GREEN else: player_data["items"] -= 1 # Thất bại trừ 1 item như yêu cầu upgrade_msg = "Thất bại! Trang bị bị tổn hại, -1 Item!" upgrade_color = RED else: upgrade_msg = "Không đủ Vàng (Cần 200) hoặc hết Item!" upgrade_color = RED # Nút Đóng bảng nâng cấp elif 375 <= mx <= 525 and 385 <= my <= 430: game_state = "PLAYING" # --- XỬ LÝ VÀ VẼ GIAO DIỆN --- if game_state == "LOGIN": # Màn hình Đăng nhập / Lưu tiến trình title = font_large.render( "ĐĂNG NHẬP TIẾN TRÌNH GAME KIẾM HIỆP", True, GOLD ) screen.blit(title, (SCREEN_WIDTH // 2 - title.get_width() // 2, 120)) # Khung nhập Tên tài khoản pygame.draw.rect( screen, WHITE if active_box == "user" else (180, 180, 180), (300, 220, 300, 40), border_radius=5, ) user_surf = font_medium.render( f"Tên: {username_input}" + ("|" if active_box == "user" else ""), True, BLACK, ) screen.blit(user_surf, (310, 228)) # Khung nhập Mật khẩu pygame.draw.rect( screen, WHITE if active_box == "pass" else (180, 180, 180), (300, 300, 300, 40), border_radius=5, ) pass_surf = font_medium.render( f"Mật khẩu: {'*' * len(password_input)}" + ("|" if active_box == "pass" else ""), True, BLACK, ) screen.blit(pass_surf, (310, 308)) # Nút bấm Đăng nhập / Chơi tiếp pygame.draw.rect(screen, RED, (350, 380, 200, 50), border_radius=8) btn_text = font_medium.render("VÀO GAME", True, WHITE) screen.blit(btn_text, (400, 393)) # Thông báo lỗi/thành công if login_message: msg_surf = font_small.render( login_message, True, GREEN if "thành công" in login_message or "mới" in login_message else RED, ) screen.blit(msg_surf, (SCREEN_WIDTH // 2 - msg_surf.get_width() // 2, 450)) # Hướng dẫn nhỏ guide = font_small.render( "Nhập tên và mật khẩu (Ví dụ: mua / 6767) để lưu & tải tiến trình chơi!", True, WHITE, ) screen.blit(guide, (SCREEN_WIDTH // 2 - guide.get_width() // 2, 500)) elif game_state == "PLAYING" or game_state == "UPGRADE": # Cập nhật chuyển động người chơi if game_state == "PLAYING": player.handle_keys() # Vẽ nền bản đồ game # Vẽ đường đi giả lập pygame.draw.rect(screen, (40, 40, 55), (0, 250, SCREEN_WIDTH, 150)) # Vẽ NPC npc.draw(screen) # Vẽ Người chơi player.draw(screen) # --- BẢNG GỖ "YOU CAN DO IT" Ở GÓC BẢN ĐỒ --- wood_rect = pygame.Rect(30, 30, 220, 80) pygame.draw.rect(screen, WOOD_BG, wood_rect, border_radius=8) pygame.draw.rect(screen, WOOD_BORDER, wood_rect, 4, border_radius=8) # Nội dung bảng gỗ sign_text1 = font_medium.render("YOU CAN DO IT!", True, GOLD) sign_text2 = font_small.render(f"Tài khoản: {current_user}", True, WHITE) screen.blit(sign_text1, (wood_rect.centerx - sign_text1.get_width() // 2, 42)) screen.blit(sign_text2, (wood_rect.centerx - sign_text2.get_width() // 2, 72)) # --- HUD THÔNG TIN NGƯỜI CHƠI (Góc trên phải) --- hud_bg = pygame.Rect(620, 25, 250, 70) pygame.draw.rect(screen, (20, 20, 25), hud_bg, border_radius=6) pygame.draw.rect(screen, GOLD, hud_bg, 2, border_radius=6) gold_surf = font_small.render(f"Vàng: {player_data['gold']} G", True, GOLD) item_surf = font_small.render( f"Số lượng Item: {player_data['items']}", True, WHITE ) screen.blit(gold_surf, (640, 38)) screen.blit(item_surf, (640, 62)) # Hướng dẫn di chuyển hint = font_small.render( "Di chuyển tới gần Lão Kiếm Sư (NPC áo đỏ phải màn hình) để nâng cấp item!", True, (180, 180, 180), ) screen.blit(hint, (20, 565)) # Nếu đang ở trạng thái UPGRADE -> Hiển thị Bảng nâng cấp item if game_state == "UPGRADE": # Lớp phủ mờ nền overlay = pygame.Surface((SCREEN_WIDTH, SCREEN_HEIGHT), pygame.SRCALPHA) overlay.fill((0, 0, 0, 150)) screen.blit(overlay, (0, 0)) # Giao diện bảng nâng cấp sắc nét phong cách cổ trang box_w, box_h = 440, 280 box_x = (SCREEN_WIDTH - res_surf.get_width() // 2, box_y + 135) ) # Nút bấm Nâng Cấp btn_up = pygame.Rect(box_x + 60, box_y + 180, 140, 40) pygame.draw.rect(screen, RED, btn_up, border_radius=6) btn_up_text = font_medium.render("NÂNG CẤP", True, WHITE) screen.blit( btn_up_text, ( btn_up.centerx -
