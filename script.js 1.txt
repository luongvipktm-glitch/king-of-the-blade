// ==========================================
// 1. KHỞI TẠO SCENE, CAMERA ISOMETRIC & RENDERER
// ==========================================
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0e0c1a);
scene.fog = new THREE.FogExp2(0x0e0c1a, 0.025);

const aspect = window.innerWidth / window.innerHeight;
const d = 10;
const camera = new THREE.OrthographicCamera(-d * aspect, d * aspect, d, -d, 1, 1000);
camera.position.set(12, 12, 12);
camera.lookAt(0, 0, 0);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
document.getElementById('game-canvas').appendChild(renderer.domElement);

// Ánh sáng môi trường
const ambientLight = new THREE.AmbientLight(0xdce0eb, 0.6);
scene.add(ambientLight);

const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
dirLight.position.set(15, 25, 15);
dirLight.castShadow = true;
scene.add(dirLight);

// Mảng lưu trữ các viên kim cương & pha lê để làm hiệu ứng xoay
const rotatingItems = [];

// ==========================================
// 2. HÀM DỰNG THẾ GIỚI 3D + KIM CƯƠNG
// ==========================================
function createWorldEnvironment() {
  const worldGroup = new THREE.Group();

  // Vật liệu (Materials)
  const stoneMat = new THREE.MeshStandardMaterial({ color: 0x5a5e6b, roughness: 0.85 });
  const darkStoneMat = new THREE.MeshStandardMaterial({ color: 0x363942, roughness: 0.9 });
  const mossMat = new THREE.MeshStandardMaterial({ color: 0x486b43, roughness: 0.95 });

  // Material Pha Lê (Xanh dương)
  const crystalMat = new THREE.MeshStandardMaterial({ 
    color: 0x00d2ff, 
    emissive: 0x0077ff, 
    emissiveIntensity: 0.8, 
    roughness: 0.1 
  });

  // Material Kim Cương (Xanh Lục Bảo / Diamond Emissive)
  const diamondMat = new THREE.MeshStandardMaterial({ 
    color: 0x00ffaa, 
    emissive: 0x00cc77, 
    emissiveIntensity: 0.9, 
    roughness: 0.05,
    metalness: 0.2
  });

  // 1. Mặt đài đá chính
  const mainFloor = new THREE.Mesh(new THREE.BoxGeometry(10, 1.5, 14), stoneMat);
  mainFloor.position.set(0, -0.75, 0);
  mainFloor.receiveShadow = true;
  worldGroup.add(mainFloor);

  // Mảng rêu & vết nứt
  const detailGeo = new THREE.BoxGeometry(1.2, 0.05, 1.2);
  const detailPositions = [
    { x: -2, z: -3, mat: mossMat }, { x: 2.5, z: 1, mat: mossMat },
    { x: -3, z: 3, mat: darkStoneMat }, { x: 1.5, z: -4, mat: darkStoneMat }
  ];
  detailPositions.forEach(item => {
    const patch = new THREE.Mesh(detailGeo, item.mat);
    patch.position.set(item.x, 0.03, item.z);
    patch.rotation.y = Math.random() * Math.PI;
    patch.receiveShadow = true;
    worldGroup.add(patch);
  });

  // Cầu đá nối
  const bridge = new THREE.Mesh(new THREE.BoxGeometry(4, 1.2, 6), darkStoneMat);
  bridge.position.set(-5, -0.9, 6);
  bridge.receiveShadow = true;
  worldGroup.add(bridge);

  // 2. Cổng vòm cổ kính
  const gateGroup = new THREE.Group();
  const leftPillar = new THREE.Mesh(new THREE.BoxGeometry(1.4, 6.5, 1.4), darkStoneMat);
  leftPillar.position.set(-3.2, 3.25, -6.5); leftPillar.castShadow = true;
  const rightPillar = new THREE.Mesh(new THREE.BoxGeometry(1.4, 6.5, 1.4), darkStoneMat);
  rightPillar.position.set(3.2, 3.25, -6.5); rightPillar.castShadow = true;
  const gateTop = new THREE.Mesh(new THREE.BoxGeometry(8, 1.5, 1.6), darkStoneMat);
  gateTop.position.set(0, 6, -6.5); gateTop.castShadow = true;

  const leftDoor = new THREE.Mesh(new THREE.BoxGeometry(2.2, 5, 0.2), stoneMat);
  leftDoor.position.set(-1.2, 2.5, -6.5);
  const rightDoor = new THREE.Mesh(new THREE.BoxGeometry(2.2, 5, 0.2), stoneMat);
  rightDoor.position.set(1.2, 2.5, -6.5);

  // Pha lê cực đại đính đỉnh cổng vòm
  const gateCrystal = new THREE.Mesh(new THREE.OctahedronGeometry(0.7), crystalMat);
  gateCrystal.position.set(0, 6, -5.6);
  rotatingItems.push(gateCrystal);

  gateGroup.add(leftPillar, rightPillar, gateTop, leftDoor, rightDoor, gateCrystal);
  worldGroup.add(gateGroup);

  // 3. Hàng cột đá
  const pillarGeo = new THREE.CylinderGeometry(0.4, 0.5, 4.5, 8);
  const pillarCapGeo = new THREE.BoxGeometry(1.1, 0.4, 1.1);
  const pillarPositions = [
    [-4.2, 2.25, -4], [-4.2, 2.25, -1], [-4.2, 2.25, 2],
    [4.2, 2.25, -1], [4.2, 2.25, 2], [4.2, 2.25, 5]
  ];
  pillarPositions.forEach(pos => {
    const pillar = new THREE.Mesh(pillarGeo, stoneMat);
    pillar.position.set(...pos); pillar.castShadow = true;
    const topCap = new THREE.Mesh(pillarCapGeo, darkStoneMat);
    topCap.position.set(pos[0], 4.4, pos[2]); topCap.castShadow = true;
    const baseCap = new THREE.Mesh(pillarCapGeo, darkStoneMat);
    baseCap.position.set(pos[0], 0.2, pos[2]);
    worldGroup.add(pillar, topCap, baseCap);
  });

  // 4. Pha lê phát sáng (Màu xanh dương)
  const crystalGeo = new THREE.OctahedronGeometry(0.4);
  const crystalSmallGeo = new THREE.OctahedronGeometry(0.25);
  const crystalClusters = [
    { x: -1.2, z: -0.5 }, { x: 1.2, z: -0.5 },
    { x: -4.5, z: -5.5 }, { x: 4.5, z: -3 }
  ];
  crystalClusters.forEach(pos => {
    const mainC = new THREE.Mesh(crystalGeo, crystalMat);
    mainC.position.set(pos.x, 0.35, pos.z);
    const subC = new THREE.Mesh(crystalSmallGeo, crystalMat);
    subC.position.set(pos.x + 0.3, 0.2, pos.z + 0.2);
    worldGroup.add(mainC, subC);
    rotatingItems.push(mainC);
  });

  // 5. THÊM KIM CƯƠNG (Màu xanh lục bảo lấp lánh)
  const diamondGeo = new THREE.OctahedronGeometry(0.35, 0); // Hình dạng kim cương vọt nhọn
  const diamondPositions = [
    { x: 0, y: 0.8, z: -3 },   // Trước cổng
    { x: -3.5, y: 0.8, z: 1 },  // Góc đài bên trái
    { x: 3.5, y: 0.8, z: 1 },   // Góc đài bên phải
    { x: -5, y: 0.8, z: 6 }     // Trên cầu đá nối
  ];

  diamondPositions.forEach(pos => {
    const diamond = new THREE.Mesh(diamondGeo, diamondMat);
    diamond.position.set(pos.x, pos.y, pos.z);
    diamond.scale.set(1, 1.3, 1); // Kéo dài để tạo hình viên kim cương đẹp mắt
    worldGroup.add(diamond);
    rotatingItems.push(diamond);
  });

  return worldGroup;
}

// ==========================================
// 3. HÀM DỰNG NHÂN VẬT (PLAYER CHARACTER)
// ==========================================
function createPlayerCharacter() {
  const playerGroup = new THREE.Group();

  const skinMat = new THREE.MeshStandardMaterial({ color: 0xe0ac69, roughness: 0.6 });
  const clothMat = new THREE.MeshStandardMaterial({ color: 0x1a1a1a, roughness: 0.9 });
  const haloMat = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.2 });
  const eyeMat = new THREE.MeshBasicMaterial({ color: 0xffffff });

  // Đầu & Mắt phát sáng
  const head = new THREE.Mesh(new THREE.BoxGeometry(1.2, 1.2, 1.2), skinMat);
  head.position.y = 3.2; head.castShadow = true;
  playerGroup.add(head);

  const leftEye = new THREE.Mesh(new THREE.BoxGeometry(0.2, 0.4, 0.05), eyeMat);
  leftEye.position.set(-0.3, 3.3, 0.61);
  const rightEye = new THREE.Mesh(new THREE.BoxGeometry(0.2, 0.4, 0.05), eyeMat);
  rightEye.position.set(0.3, 3.3, 0.61);
  playerGroup.add(leftEye, rightEye);

  // Vòng Halo
  const halo = new THREE.Mesh(new THREE.TorusGeometry(0.8, 0.08, 16, 100), haloMat);
  halo.rotation.x = Math.PI / 2;
  halo.position.y = 4.2;
  playerGroup.add(halo);

  // Thân áo đen
  const body = new THREE.Mesh(new THREE.BoxGeometry(1.4, 1.5, 0.9), clothMat);
  body.position.y = 1.9; body.castShadow = true;
  playerGroup.add(body);

  // Tay & Chân
  const leftArm = new THREE.Mesh(new THREE.BoxGeometry(0.4, 1.3, 0.5), skinMat);
  leftArm.position.set(-0.95, 1.9, 0); leftArm.castShadow = true;
  const rightArm = new THREE.Mesh(new THREE.BoxGeometry(0.4, 1.3, 0.5), skinMat);
  rightArm.position.set(0.95, 1.9, 0); rightArm.castShadow = true;
  playerGroup.add(leftArm, rightArm);

  const leftLeg = new THREE.Mesh(new THREE.BoxGeometry(0.5, 1.2, 0.5), skinMat);
  leftLeg.position.set(-0.35, 0.6, 0); leftLeg.castShadow = true;
  const rightLeg = new THREE.Mesh(new THREE.BoxGeometry(0.5, 1.2, 0.5), skinMat);
  rightLeg.position.set(0.35, 0.6, 0); rightLeg.castShadow = true;
  playerGroup.add(leftLeg, rightLeg);

  return playerGroup;
}

// ==========================================
// 4. HIỂN THỊ THẾ GIỚI, NHÂN VẬT & VÒNG LẶP ANIMATE
// ==========================================
const world = createWorldEnvironment();
scene.add(world);

const player = createPlayerCharacter();
player.position.set(0, 0, 2); // Đặt nhân vật đứng giữa đài
scene.add(player);

// Vòng lặp render màn hình & tạo chuyển động xoay kim cương
function animate() {
  requestAnimationFrame(animate);

  // Hiệu ứng xoay kim cương và pha lê liên tục
  rotatingItems.forEach(item => {
    item.rotation.y += 0.02;
  });

  renderer.render(scene, camera);
}
animate();
